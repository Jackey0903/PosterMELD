\newcommand{\Stage}[7]{%
  \node[stage,draw=#7!48] (#1) at (#2,#3) {};
  \fill[#7] ([xshift=0.02cm,yshift=-0.02cm]#1.north west)
    rectangle ([xshift=0.12cm,yshift=0.02cm]#1.south west);
  \node[number,fill=#7] at ([xshift=0.48cm,yshift=0.32cm]#1.west) {#4};
  \node[anchor=west,text=Ink,font=\bfseries\fontsize{11}{13}\selectfont]
    at ([xshift=0.90cm,yshift=0.32cm]#1.west) {#5};
  \draw[Border,line width=0.45pt]
    ([xshift=0.34cm,yshift=-0.06cm]#1.west) --
    ([xshift=-0.34cm,yshift=-0.06cm]#1.east);
  \node[anchor=west,text=Muted,font=\fontsize{7.7}{9.2}\selectfont]
    at ([xshift=0.34cm,yshift=-0.45cm]#1.west) {#6};
}
